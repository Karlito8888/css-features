@keyframes bounce {

  50% { translate: 0 -48px; animation-timing-function: ease-in; }
}

.note { --note: "0%, 50% and 100%: up and down, with an easing set per keyframe"; }

@keyframes traffic {

  35% { background: var(--a4); }
  70% { background: var(--a2); }
}

@keyframes grow {

}
